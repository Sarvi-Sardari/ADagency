# Python runtime for the ad_agency branch on Linux and Python 3.12.
# Covers planner generation, anomaly detection/rerun, mask generation, XAI,
# metrics, hardware monitoring, and local or Ray/vLLM cluster execution.
#
# Fresh Conda environment (run from the pipeline_project directory):
#   conda create -n xai python=3.12 pip -y
#   conda activate xai
#   python -m pip install --upgrade pip
#
# Install the CUDA 13.0 development toolkit needed to compile FlashAttention:
#   conda install -c nvidia cuda-toolkit=13.0 -y
# Verify the NVIDIA driver and compiler before installing Python packages:
#   nvidia-smi
#   nvcc --version
#
# FlashAttention is compiled against PyTorch, so install the pinned PyTorch
# build first and then install this complete file without build isolation:
#   python -m pip install torch==2.13.0 torchvision==0.28.0
#   python -m pip install --no-build-isolation -r requirements.txt
#
# Windows support:
#   - Run the complete pipeline inside WSL2; native Windows is not supported by
#     FlashAttention, vLLM, or this pipeline's Linux /proc telemetry collector.
#   - From an elevated Windows PowerShell, install WSL once with:
#       wsl --install -d Ubuntu-24.04
#   - Then run the Conda/CUDA/Python commands above inside the WSL terminal.
#
# TensorRT-LLM is not supported: it pins torch/transformers versions that
# conflict with this file, was not faster than vLLM in a benchmark, and does
# not expose the per-token log-probabilities used for step confidence.
#
# Other system prerequisites that pip cannot install:
#   - Install a current NVIDIA driver on the host before the CUDA command above.
#   - SSH keys/connectivity are needed only when deployment.mode is "cluster".
#
# Model assets that pip cannot place in this project:
#   - Download the VLM/LLM checkpoints selected in configs/model_configs.yaml.
#   - SAM3 source expected by the default configs:
#       git clone https://github.com/facebookresearch/sam3.git models/sam3
#       git -C models/sam3 checkout 11dec2936de97f2857c1f76b66d982d5a001155d
#   - SAM3 is gated on Hugging Face. After access is approved:
#       hf auth login
#       hf download facebook/sam3 sam3.pt --local-dir models/sam3_model
